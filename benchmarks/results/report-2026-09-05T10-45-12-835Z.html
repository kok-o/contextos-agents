import crypto from 'node:crypto';

const scryptAsync = promisify(crypto.scrypt);
const timingSafeEqual = crypto.timingSafeEqual;

export async function verifyPassword(password: string, storedHash: string, salt: string): Promise<boolean> {
  const hashBuffer = Buffer.from(storedHash, 'hex');
  const derivedKey = (await scryptAsync(password, salt, 64)) as Buffer;
  const keyBuffer = Buffer.from(derivedKey); // already Buffer
  if (hashBuffer.length !== keyBuffer.length) return false;
  return timingSafeEqual(hashBuffer, keyBuffer);
}

private async checkAndRecordFailure(key: string): Promise<void> {
  const record = await this.rateLimitStore.get(key);
  const now = Date.now();

  if (record?.blockedUntil && record.blockedUntil > now) {
    throw new RateLimitError(Math.ceil((record.blockedUntil - now)/1000));
  }

  // If lockout expired and record exists, reset failures to 0? Actually if expired, we still have failures from before. If next failure should double lockout? We can keep failure count to double. But if lockout expired, maybe old failures should be considered? "exponential lockout" means after expiry, if another attempt fails, lockout increases. Keep failures. If no record, new record.
  const failures = (record?.blockedUntil && record.blockedUntil <= now ? record.failures : record?.failures ?? 0);
  const nextFailures = failures + 1;
  let blockedUntil: number | null = null;
  if (nextFailures >= MAX_FAILURES) {
    const exponent = nextFailures - MAX_FAILURES;
    const delayMs = Math.min(LOCKOUT_BASE_MS * 2 ** exponent, LOCKOUT_MAX_MS);
    blockedUntil = now + delayMs;
  }
  await this.rateLimitStore.set(key, { failures: nextFailures, blockedUntil });
